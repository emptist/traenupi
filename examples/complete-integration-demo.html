🔀 Hybrid Routing Strategy

How the Bridge Routes Requests

Path/Key Backend Reason
/tmp/file.txt Lively4 Local temporary file
/cache/data.json Lively4 Local cache
/documents/report.txt Puter Cloud document
/projects/code.js Puter Cloud project file
cache:session Local DB Fast local access
user:preferences Cloud DB Persistent user data

💻 Integration Examples

// Using Bridge in TraeNuPI TypeScript
import { LivelyPuterBridge } from './src/common/lively-puter/index.js';

export class MyTraeNuPIModule {
  private bridge: LivelyPuterBridge;
  
  async initialize() {
    this.bridge = new LivelyPuterBridge({
      mode: 'hybrid',  // Auto-routing
      enableAI: true,
      enableStorage: true
    });
    await this.bridge.initialize();
  }
  
  async saveData(data: any) {
    // Automatically routes to best backend
    await this.bridge.db.set('data:my-module', data);
  }
  
  async analyzeWithAI(text: string) {
    const response = await this.bridge.ai.chat([
      { role: 'user', content: text }
    ]);
    return response.message.content;
  }
}
// Using Bridge in Gleam
import traenupi_core/lively_puter as bridge
import gleam/json

pub fn use_bridge() {
  // Initialize
  let config = bridge.default_config()
  let assert Ok(_) = bridge.initialize(config)
  
  // File operations
  let assert Ok(_) = bridge.write_file("/test.txt", "Hello!")
  let assert Ok(content) = bridge.read_file("/test.txt")
  
  // AI chat
  let messages = [bridge.UserMessage("Hello!")]
  let assert Ok(response) = bridge.chat(messages)
  
  // Database
  let value = json.object([#("key", json.string("value"))])
  let assert Ok(_) = bridge.db_set("my-key", value)
}
// Hybrid Mode Routing Example
const bridge = new LivelyPuterBridge({ mode: 'hybrid' });
await bridge.initialize();

// File System Routing
await bridge.fs.write('/tmp/cache.json', data);     // → Lively4 (local)
await bridge.fs.write('/documents/report.txt', text); // → Puter (cloud)

// Database Routing
await bridge.db.set('cache:session', session);      // → Local (fast)
await bridge.db.set('user:preferences', prefs);    // → Cloud (persistent)

📊 Integration Summary

✅ What's Integrated

  • TraeNuPI: Bridge is part of core, available in all modules
  • Lively4: Local services, hybrid routing, real-time collaboration
  • Puter: Cloud services, AI, database, authentication
  • Bridge: Unified API, intelligent routing, type-safe
  • Gleam: Full FFI bindings for functional programming

🎯 Key Benefits

  • Zero Configuration: Auto-detect best mode
  • Performance: Local caching + cloud persistence
  • Reliability: Automatic fallback mechanisms
  • Type Safety: Full TypeScript and Gleam support
  • Flexibility: Three modes: lively-first, puter-first, hybrid
